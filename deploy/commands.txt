# Deployment :
 
vi deployhttpd1.yml
 
kind: Deployment
apiVersion: apps/v1
metadata:
  name: mydeployment
spec:
  replicas: 2
  selector:
    matchLabels:
      name: deployment1
  template:
    metadata:
      name: testp1
      labels:
        name: deployment1
    spec:
      containers:
        - name: c00
          image: httpd
          ports:
            - containerPort: 80
 
 
kubectl apply -f deployhttpd1.yml
 
 
## Access the app in the pod from the load balancer
 
# vi service.yml
 
kind: Service
apiVersion: v1
metadata:
  name: demoservices
spec:
  ports:
    - port: 8080
      targetPort: 80
  selector:
    name: deployment1
  type: LoadBalancer


# vi pod.yml

kind: Pod
apiVersion: v1

  name: myfirstpod
